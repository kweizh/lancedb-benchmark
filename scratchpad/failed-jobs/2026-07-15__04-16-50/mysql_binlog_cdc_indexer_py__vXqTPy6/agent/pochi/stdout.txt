import hashlib, math

def get_vector(title, body):
    title = title or ''
    body = body or ''
    text = title + ' ' + body
    vector = [0.0] * 32
    tokens = text.lower().split()
    for token in tokens:
        bucket = int(hashlib.md5(token.encode('utf-8')).hexdigest(), 16) % 32
        vector[bucket] += 1.0
    norm = math.sqrt(sum(x * x for x in vector))
    if norm > 0:
        vector = [x / norm for x in vector]
    return vector

print(get_vector('Hello', 'World'))
print(get_vector('', ''))

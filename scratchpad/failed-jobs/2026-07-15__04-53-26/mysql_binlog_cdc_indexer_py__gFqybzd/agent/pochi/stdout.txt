def ref_vec(title, body):
    text = ("" if title is None else str(title)) + " " + ("" if body is None else str(body))
    vec = [0.0] * 32
    for token in text.lower().split():
        bucket = int(hashlib.md5(token.encode("utf-8")).hexdigest(), 16) % 32
        vec[bucket] += 1.0
    norm = math.sqrt(sum(v*v for v in vec))
    if norm > 0.0:
        vec = [v/norm for v in vec]
    return vec

    text = ("" if title is None else str(title)) + " " + ("" if body is None else str(body))
    vec = [0.0] * 32
    for token in text.lower().split():
        bucket = int(hashlib.md5(token.encode("utf-8")).hexdigest(), 16) % 32
        vec[bucket] += 1.0

    return vec

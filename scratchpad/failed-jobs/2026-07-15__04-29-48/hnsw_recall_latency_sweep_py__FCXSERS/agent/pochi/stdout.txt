import lancedb
import numpy as np
import time

db = lancedb.connect('/app/lancedb')
tbl = db.open_table('vectors_test123')

rng = np.random.default_rng(42)
queries = rng.standard_normal((20, 64)).astype('float32')

# Compute ground truth for all queries
gt_ids_list = []
for q in queries:
    bf = (tbl.search(q.tolist(), vector_column_name='vector')
          .metric('l2')
          .bypass_vector_index()
          .limit(50)
          .to_list())
    bf_sorted = sorted(bf, key=lambda r: (r['_distance'], r['id']))
    gt_ids_list.append([r['id'] for r in bf_sorted[:10]])

# Test recall for various configs
for np_ in [1, 8, 32, 64]:
    for rf in [1, 5, 10, 20, 50]:
        recalls = []
        for i, q in enumerate(queries):
            ann = (tbl.search(q.tolist(), vector_column_name='vector')
                   .metric('l2')
                   .nprobes(np_)
                   .refine_factor(rf)
                   .limit(10)
                   .to_list())
            ann_ids = set(r['id'] for r in ann[:10])
            gt_ids = set(gt_ids_list[i])
            recall = len(ann_ids & gt_ids) / 10
            recalls.append(recall)
        avg_recall = np.mean(recalls)
        print(f'nprobes={np_}, refine={rf}: avg_recall={avg_recall:.3f}')
